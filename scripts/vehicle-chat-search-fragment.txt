    const q = raw.toUpperCase().replace(/[-\s]/g, '');

    // VEHICLE_CHAT_SEARCH_FIX_20260714
    // Natural-language vehicle lookup used by the public chatbot and /api/search.
    // Example: "¿Qué filtros usa un Toyota Corolla 2017?"
    // VIN decoding remains outside this flow and can continue to use NHTSA later.
    const vehicleYearMatch = raw.match(/\b(19\d{2}|20\d{2})\b/);
    const vehicleStopWords = new Set([
      'A','AN','AND','ARE','CAR','CARS','DO','DOES','EL','EN','FILTER','FILTERS','FILTRO','FILTROS',
      'FOR','HOW','LA','LAS','LOS','ME','MI','NEED','PARA','POR','QUE','QUÉ','THE','THIS','UN','UNA',
      'USA','USE','USES','USO','USAR','UTILIZA','UTILIZAN','VEHICLE','VEHICULO','VEHÍCULO','WHAT','WHICH'
    ]);
    const vehicleTokens = raw
      .normalize('NFKD')
      .replace(/[\u0300-\u036f]/g, '')
      .toUpperCase()
      .replace(/\b(19\d{2}|20\d{2})\b/g, ' ')
      .replace(/[^A-Z0-9]+/g, ' ')
      .trim()
      .split(/\s+/)
      .filter(Boolean)
      .filter(token => token.length >= 2 && !vehicleStopWords.has(token));

    const looksLikeVehicleQuestion = Boolean(vehicleYearMatch) && vehicleTokens.length >= 2 && /[A-Za-zÁÉÍÓÚÜÑáéíóúüñ]/.test(raw);

    if (looksLikeVehicleQuestion) {
      const requestedYear = Number(vehicleYearMatch[1]);
      const uniqueTokens = [...new Set(vehicleTokens)].slice(0, 5);
      const makeToken = uniqueTokens[0];
      const modelToken = uniqueTokens[1] || '';

      let vehicleRows;
      try {
        vehicleRows = await client.query(
          `SELECT c.*
           FROM elimfilters_catalog c
           WHERE c.duty = 'LIGHT_DUTY'
             AND c.vehicle_applications IS NOT NULL
             AND jsonb_typeof(c.vehicle_applications) = 'array'
             AND c.vehicle_applications @> jsonb_build_array(jsonb_build_object('make', $1::text))
             AND EXISTS (
               SELECT 1
               FROM jsonb_array_elements(c.vehicle_applications) AS va
               WHERE UPPER(
                 COALESCE(va->>'model','') || ' ' ||
                 COALESCE(va->>'model_family','') || ' ' ||
                 COALESCE(va->>'model_type','')
               ) LIKE $2::text
             )
           ORDER BY c.sku
           LIMIT 120`,
          [makeToken, '%' + modelToken + '%']
        );
      } catch (vehicleSearchError) {
        console.error('[search:vehicle]', vehicleSearchError.code || '', vehicleSearchError.message);
        return res.status(500).json({
          success: false,
          error: 'Vehicle search failed',
          code: vehicleSearchError.code || null,
          detail: vehicleSearchError.message
        });
      }

      const expandYear = (value) => {
        const n = Number(value);
        if (!Number.isFinite(n)) return null;
        if (String(value).length === 4) return n;
        return n <= 35 ? 2000 + n : 1900 + n;
      };

      const applicationCoversYear = (application, year) => {
        if (!application || typeof application !== 'object') return false;
        const rangeText = String(application.year_range || '').trim();
        const yearText = String(application.year || '').trim();
        const modelText = String(application.model || '');

        const usaExact = modelText.match(/\(USA\)\s*(19\d{2}|20\d{2})/i);
        if (usaExact) return Number(usaExact[1]) === year;

        const sourceText = rangeText || yearText;
        if (!sourceText) return true;

        const datedYears = [...sourceText.matchAll(/\b\d{1,2}\/(\d{2}|\d{4})\b/g)]
          .map(match => expandYear(match[1]))
          .filter(Boolean);

        if (datedYears.length) {
          const start = datedYears[0];
          const hasRange = sourceText.includes('→') || sourceText.includes('->');
          const end = hasRange && datedYears.length > 1 ? datedYears[datedYears.length - 1] : null;
          return year >= start && (end === null || year <= end);
        }

        const fullYears = [...sourceText.matchAll(/\b(19\d{2}|20\d{2})\b/g)]
          .map(match => Number(match[1]));
        if (!fullYears.length) return true;
        if (fullYears.length === 1) return fullYears[0] === year;
        return year >= fullYears[0] && year <= fullYears[fullYears.length - 1];
      };

      const matchesTokens = (application) => {
        const haystack = [
          application?.make,
          application?.model,
          application?.model_family,
          application?.model_type
        ].filter(Boolean).join(' ').normalize('NFKD').replace(/[\u0300-\u036f]/g, '').toUpperCase();
        return uniqueTokens.every(token => haystack.includes(token));
      };

      const categoryLabels = lang === 'es'
        ? { cabin: 'Cabina', oil: 'Aceite', air: 'Aire de motor', fuel: 'Combustible', coolant: 'Refrigerante', transmission: 'Transmisión', other: 'Otros' }
        : { cabin: 'Cabin', oil: 'Oil', air: 'Engine Air', fuel: 'Fuel', coolant: 'Coolant', transmission: 'Transmission', other: 'Other' };

      const classifyVehicleFilter = (product) => {
        const text = [
          product.description,
          product.filter_type,
          product.filter_subtype,
          product.installation_type,
          product.technology,
          product.elimfilters_sku
        ].filter(Boolean).join(' ').normalize('NFKD').replace(/[\u0300-\u036f]/g, '').toUpperCase();
        const sku = String(product.elimfilters_sku || '').toUpperCase();

        if (/\b(CABIN|POLLEN|INTERIOR AIR|AIR CONDITIONING|HVAC|HABITACLE)\b/.test(text) || sku.startsWith('EC')) return 'cabin';
        if (/\b(OIL|LUBE|LUBRICATION|ENGINE OIL)\b/.test(text) || sku.startsWith('EL')) return 'oil';
        if (/\b(FUEL|DIESEL|GASOLINE|PETROL|WATER SEPARATOR)\b/.test(text) || sku.startsWith('EF')) return 'fuel';
        if (/\b(COOLANT|COOLING|ANTIFREEZE)\b/.test(text)) return 'coolant';
        if (/\b(TRANSMISSION|GEARBOX|ATF|CVT)\b/.test(text)) return 'transmission';
        if (/\b(AIR FILTER|ENGINE AIR|AIR INTAKE|INTAKE AIR)\b/.test(text) || sku.startsWith('EA')) return 'air';
        return 'other';
      };

      const vehicleProducts = vehicleRows.rows.map(row => {
        const product = buildFilterData(row, lang);
        const applications = Array.isArray(row.vehicle_applications) ? row.vehicle_applications : [];
        const matchedApplications = applications
          .filter(app => matchesTokens(app) && applicationCoversYear(app, requestedYear))
          .slice(0, 25);
        if (!matchedApplications.length) return null;

        const category = classifyVehicleFilter(product);
        return {
          sku: product.elimfilters_sku,
          category,
          category_label: categoryLabels[category],
          technology: product.technology || null,
          filter_type: product.filter_type || null,
          applications: matchedApplications.map(app => ({
            make: app.make || null,
            model: app.model || app.model_family || null,
            model_type: app.model_type || null,
            engine: app.engine_code || app.engine || null,
            year_range: app.year_range || app.year || null,
            ccm: app.ccm || null,
            hp: app.hp || null,
            kw: app.kw || null
          }))
        };
      }).filter(Boolean);

      if (vehicleProducts.length > 0) {
        const engineOptions = [...new Set(vehicleProducts.flatMap(product =>
          product.applications.map(app => app.engine || app.model_type || '')
        ).filter(Boolean))].slice(0, 20);

        const categoryOrder = ['oil', 'air', 'cabin', 'fuel', 'coolant', 'transmission', 'other'];
        const categories = categoryOrder
          .map(key => ({
            key,
            label: categoryLabels[key],
            products: vehicleProducts.filter(product => product.category === key)
          }))
          .filter(group => group.products.length > 0);

        return res.json({
          success: true,
          source: 'vehicle_natural_language_compact',
          vehicle: {
            make: makeToken,
            model: modelToken,
            year: requestedYear
          },
          needs_engine_confirmation: engineOptions.length > 1,
          engine_options: engineOptions,
          categories,
          total_products: vehicleProducts.length
        });
      }

      return res.json({
        success: true,
        source: 'vehicle_no_match',
        vehicle: { make: makeToken, model: modelToken, year: requestedYear },
        needs_engine_confirmation: false,
        engine_options: [],
        categories: [],
        total_products: 0
      });
    }
